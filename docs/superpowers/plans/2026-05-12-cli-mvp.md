# Scrip CLI MVP — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** CLI Node.js qui prend une photo de cours manuscrit et produit un fichier Markdown structuré via Mistral OCR + Gemma 4 (OpenRouter).

**Architecture:** `ingest.js` orchestre le pipeline en trois étapes : (1) OCR via Mistral pour extraire le texte brut, (2) structuration via Gemma 4 sur OpenRouter pour produire du Markdown propre, (3) ajout du frontmatter YAML et écriture stdout. Chaque étape est un module isolé avec une interface claire.

**Tech Stack:** Node.js 20+ (ESM natif, `fetch` natif, `node:test` pour les tests), dotenv pour les clés API. Zéro dépendance runtime supplémentaire.

```js
// src/format.js

/**
 * Assemble un fichier Markdown Obsidian avec frontmatter YAML.
 * @param {string} content - Corps markdown structuré (sans frontmatter)
 * @param {{ subject: string, level: string, dateCours: string, dateIngestion: string, sources: string[] }} meta
 * @returns {string}
 */
export function buildNote(content, meta) {
  const sourcesInline = meta.sources.join(', ');
  const frontmatter = [
    '---',
    `matière: ${meta.subject}`,
    `niveau: ${meta.level}`,
    `date_cours: ${meta.dateCours}`,
    `date_ingestion: ${meta.dateIngestion}`,
    `tags: [cours, ${meta.subject}, ${meta.level.replace(/\s+/g, '-')}]`,
    `sources: [${sourcesInline}]`,
    '---',
  ].join('\n');

  return `${frontmatter}\n\n${content}\n`;
}
```

```js
// src/ocr.js
import fs from 'node:fs';
import path from 'node:path';

const MISTRAL_OCR_ENDPOINT = 'https://api.mistral.ai/v1/ocr';

/**
 * Envoie une image à Mistral OCR et retourne le texte brut + les régions d'images détectées.
 * @param {string} imagePath - Chemin absolu ou relatif vers le fichier image
 * @param {string} apiKey - Clé API Mistral
 * @returns {Promise<{ rawMarkdown: string, imageRegions: Array<{id:string, x1:number, y1:number, x2:number, y2:number}> }>}
 */
export async function runOCR(imagePath, apiKey) {
  const imageBuffer = fs.readFileSync(imagePath);
  const base64 = imageBuffer.toString('base64');
  const ext = path.extname(imagePath).toLowerCase().replace('.', '');
  const mimeType = ext === 'png' ? 'image/png' : 'image/jpeg';

  const response = await fetch(MISTRAL_OCR_ENDPOINT, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${apiKey}`,
    },
    body: JSON.stringify({
      model: 'mistral-ocr-latest',
      document: {
        type: 'image_url',
        image_url: `data:${mimeType};base64,${base64}`,
      },
    }),
  });

  if (!response.ok) {
    const errorText = await response.text();
    throw new Error(`Mistral OCR erreur ${response.status}: ${errorText}`);
  }

  const data = await response.json();

  // Concatène les pages (une seule pour une photo de cours)
  const rawMarkdown = data.pages.map((p) => p.markdown).join('\n\n');

  const imageRegions = data.pages.flatMap((p) =>
    (p.images ?? []).map((img) => ({
      id: img.id,
      x1: img.top_left_x,
      y1: img.top_left_y,
      x2: img.bottom_right_x,
      y2: img.bottom_right_y,
    }))
  );

  return { rawMarkdown, imageRegions };
}
```

```js
// src/structure.js
import { HISTOIRE_SYSTEM_PROMPT, HISTOIRE_USER_TEMPLATE } from '../prompts/histoire.js';

const OPENROUTER_ENDPOINT = 'https://openrouter.ai/api/v1/chat/completions';
const PRIMARY_MODEL = 'google/gemma-4-26b-a4b-it:free';

const PROMPTS = {
  histoire: { system: HISTOIRE_SYSTEM_PROMPT, userTemplate: HISTOIRE_USER_TEMPLATE },
};

/**
 * Envoie le texte OCR brut à Gemma 4 via OpenRouter pour structuration markdown.
 * @param {string} rawMarkdown - Sortie brute de l'OCR
 * @param {string} subject - Matière (ex: 'histoire')
 * @param {string} level - Niveau scolaire (ex: '1ère')
 * @param {string} apiKey - Clé API OpenRouter
 * @returns {Promise<string>} Markdown structuré
 */
export async function structureContent(rawMarkdown, subject, level, apiKey) {
  const prompt = PROMPTS[subject] ?? PROMPTS['histoire'];

  const response = await fetch(OPENROUTER_ENDPOINT, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${apiKey}`,
      'HTTP-Referer': 'https://scrip.local',
      'X-Title': 'Scrip',
    },
    body: JSON.stringify({
      model: PRIMARY_MODEL,
      temperature: 0.2,
      messages: [
        { role: 'system', content: prompt.system },
        { role: 'user', content: prompt.userTemplate(rawMarkdown) },
      ],
    }),
  });

  if (!response.ok) {
    const errorText = await response.text();
    throw new Error(`OpenRouter erreur ${response.status}: ${errorText}`);
  }

  const data = await response.json();
  return data.choices[0].message.content;
}
```

```js
#!/usr/bin/env node
// ingest.js
import 'dotenv/config';
import path from 'node:path';
import { runOCR } from './src/ocr.js';
import { structureContent } from './src/structure.js';
import { buildNote } from './src/format.js';

const SUBJECT = 'histoire';
const LEVEL = process.env.SCRIP_LEVEL ?? '1ère';

async function main() {
  const imagePath = process.argv[2];

  if (!imagePath) {
    console.error('Usage: node ingest.js <chemin_image.jpg>');
    process.exit(1);
  }

  const mistralKey = process.env.MISTRAL_API_KEY;
  const openrouterKey = process.env.OPENROUTER_API_KEY;

  if (!mistralKey || !openrouterKey) {
    console.error('Erreur : MISTRAL_API_KEY et OPENROUTER_API_KEY doivent être définis dans .env');
    process.exit(1);
  }

  const absolutePath = path.resolve(imagePath);
  const sourceName = path.basename(imagePath);

  process.stderr.write(`[1/3] OCR en cours sur ${sourceName}...\n`);
  const { rawMarkdown } = await runOCR(absolutePath, mistralKey);

  process.stderr.write(`[2/3] Structuration via Gemma 4...\n`);
  const structuredContent = await structureContent(rawMarkdown, SUBJECT, LEVEL, openrouterKey);

  process.stderr.write(`[3/3] Assemblage de la note...\n`);
  const now = new Date();
  const dateCours = now.toISOString().split('T')[0];
  const dateIngestion = now.toISOString();

  const note = buildNote(structuredContent, {
    subject: SUBJECT,
    level: LEVEL,
    dateCours,
    dateIngestion,
    sources: [sourceName],
  });

  process.stdout.write(note);
}

main().catch((err) => {
  console.error('Erreur pipeline :', err.message);
  process.exit(1);
});
```
