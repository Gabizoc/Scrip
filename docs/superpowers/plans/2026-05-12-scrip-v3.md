```js
// prompts/shs.js

export const SHS_SYSTEM_PROMPT = `Tu es un assistant spécialisé dans la structuration de cours de lycée français (Histoire, Géographie, EMC).

Tu reçois le texte brut extrait par OCR d'un cours manuscrit. Produis un fichier Markdown propre, structuré selon les conventions pédagogiques françaises, prêt à intégrer un vault Obsidian.

## Hiérarchie des titres

- \`#\` (H1) : titre du cours (détecté ou inféré du contenu)
- \`##\` (H2) : grandes parties en chiffres romains — \`## I- Titre de la partie\`
- \`###\` (H3) : sous-parties lettrées — \`### A) Titre de la sous-partie\`
- \`####\` (H4) si nécessaire — \`#### 1. Point\`
- \`**Introduction**\` et \`**Problématique**\` : en gras sur leur propre ligne, jamais en titre H2
- \`***\` seul sur une ligne : séparateur entre les grandes parties, avant la conclusion, et avant la section de définitions

## Listes

Style français obligatoire : chaque puce se termine par \`;\` sauf la dernière qui se termine par \`.\`
Utilise les listes uniquement pour des éléments réellement énumérés dans le cours manuscrit.

## Tableaux

Utilise un tableau Markdown quand le contenu compare des éléments typologiques (territoires, milieux, périodes, acteurs). Format : \`| Col | Col |\` avec ligne \`|---|---|\`.

## Mise en forme sémantique

- Termes-clés du programme : \`**terme**\` en gras à leur première occurrence uniquement.
- Définitions : inline \`**terme** : définition.\` ou section finale \`## Définitions (à apprendre)\` si ≥ 3 définitions.

## Politique de correction

### Corrections silencieuses — sans mention dans la sortie
- Orthographe, grammaire, accords, conjugaisons, syntaxe de l'élève.
- Mots coupés en fin de ligne par tiret : recoller automatiquement. Exemple : "Fontaine-Fran-" en fin de ligne + "çaise" en début de ligne suivante → "Fontaine-Française".
- Ponctuation manquante ou incorrecte.
- Reformulation légère pour fluidité, sens exact conservé.

### Corrections tracées — mention obligatoire dans la sortie
Uniquement pour les erreurs factuelles vérifiables : dates aberrantes, noms propres historiques erronés, événements mal datés.
Format exact : \`[Note : "<texte original OCR>" corrigé en "<texte corrigé>" — <raison concise>]\`
Exemple : \`[Note : "4589" corrigé en "1589" — date du coup d'État du 18 Brumaire]\`

### Politique d'incertitude stricte — RÈGLE ABSOLUE
- Terme illisible sans hypothèse possible : \`[illisible : "<ce que l'OCR a produit>"]\`
- Hypothèse contextuelle incertaine : \`<hypothèse> [incertain — OCR : "<texte brut OCR>"]\`
- INTERDIT ABSOLU : substituer un mot par un autre "plausible" sans signal d'incertitude visible dans la sortie. Si le mot OCR ne correspond à rien de reconnaissable, utilise l'une des deux formes ci-dessus.

## Bloc METADATA — OBLIGATOIRE

Termine TOUJOURS ta réponse par ce bloc HTML commenté. L'évaluation porte sur le texte OCR brut reçu, avant tes corrections silencieuses.

<!-- METADATA
qualite_francais: <note sur 100>
nb_fautes_orthographe: <entier>
nb_fautes_grammaire: <entier>
nb_fautes_syntaxe: <entier>
commentaire: "<une phrase synthétique, max 100 caractères>"
-->

## Format de sortie

Markdown pur uniquement. Pas de bloc de code. Pas d'introduction ni de conclusion de ta part. Commence directement par \`# Titre du cours\`. Le bloc METADATA est la dernière chose dans ta réponse.`;

export const SHS_USER_TEMPLATE = (ocrText) =>
  `Voici le texte OCR brut d'un cours. Structure-le en markdown selon les instructions :\n\n${ocrText}`;
```

```js
// src/structure.js
import { SHS_SYSTEM_PROMPT, SHS_USER_TEMPLATE } from '../prompts/shs.js';

const MISTRAL_ENDPOINT = 'https://api.mistral.ai/v1/chat/completions';
const OPENROUTER_ENDPOINT = 'https://openrouter.ai/api/v1/chat/completions';

const OPENROUTER_FALLBACKS = [
  'google/gemma-4-31b-it:free',
  'nvidia/nemotron-3-super-120b-a12b:free',
  'qwen/qwen3-next-80b-a3b-instruct:free',
  'openrouter/auto',
];

const PROMPTS = {
  histoire:    { system: SHS_SYSTEM_PROMPT, userTemplate: SHS_USER_TEMPLATE },
  géographie:  { system: SHS_SYSTEM_PROMPT, userTemplate: SHS_USER_TEMPLATE },
  emc:         { system: SHS_SYSTEM_PROMPT, userTemplate: SHS_USER_TEMPLATE },
};

async function callMistral(messages, apiKey) {
  const response = await fetch(MISTRAL_ENDPOINT, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${apiKey}`,
    },
    body: JSON.stringify({ model: 'mistral-small-latest', temperature: 0.2, messages }),
  });

  if (response.status === 429) return null;

  if (!response.ok) {
    const errorText = await response.text();
    throw new Error(`Mistral erreur ${response.status}: ${errorText}`);
  }

  const data = await response.json();
  return data.choices[0].message.content;
}

async function callOpenRouter(messages, apiKey, model) {
  const response = await fetch(OPENROUTER_ENDPOINT, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${apiKey}`,
      'HTTP-Referer': 'https://scrip.local',
      'X-Title': 'Scrip',
    },
    body: JSON.stringify({ model, temperature: 0.2, messages }),
  });

  if (response.status === 429) return null;

  if (!response.ok) {
    const errorText = await response.text();
    throw new Error(`OpenRouter erreur ${response.status}: ${errorText}`);
  }

  const data = await response.json();
  return data.choices[0].message.content;
}

/**
 * Envoie le texte OCR brut au LLM pour structuration markdown.
 * Essaie d'abord Mistral direct, puis la chaîne OpenRouter sur 429.
 * @param {string} rawMarkdown - Sortie brute de l'OCR
 * @param {string} subject - Matière ('histoire' | 'géographie' | 'emc')
 * @param {string} level - Niveau scolaire (ex: '1ère')
 * @param {string} openrouterKey - Clé API OpenRouter
 * @param {string} mistralKey - Clé API Mistral
 * @returns {Promise<string>} Markdown structuré
 */
export async function structureContent(rawMarkdown, subject, level, openrouterKey, mistralKey) {
  const prompt = PROMPTS[subject] ?? PROMPTS['histoire'];
  const messages = [
    { role: 'system', content: prompt.system },
    { role: 'user', content: prompt.userTemplate(rawMarkdown) },
  ];

  process.stderr.write(`    → mistral-small-latest (direct)\n`);
  const mistralResult = await callMistral(messages, mistralKey);
  if (mistralResult !== null) return mistralResult;
  process.stderr.write(`    ⚠ 429 sur Mistral, bascule OpenRouter...\n`);

  for (const model of OPENROUTER_FALLBACKS) {
    process.stderr.write(`    → ${model}\n`);
    const result = await callOpenRouter(messages, openrouterKey, model);
    if (result !== null) return result;
    process.stderr.write(`    ⚠ 429 sur ${model}, essai suivant...\n`);
  }

  throw new Error('Tous les modèles LLM sont rate-limited. Réessaie dans quelques minutes.');
}
```

```js
// src/audit.js
import fs from 'node:fs';
import path from 'node:path';

const CORRECTION_RE = /\[Note\s*:\s*"([^"]+)"\s+corrigé\s+en\s+"([^"]+)"\s*[—–-]\s*([^\]]+)\]/gu;
const AUDIT_DIR = '_audit';

/**
 * Parse les corrections factuelles tracées du contenu LLM.
 * @param {string} content
 * @returns {Array<{ original: string, corrected: string, reason: string }>}
 */
export function parseCorrections(content) {
  const results = [];
  let match;
  const re = new RegExp(CORRECTION_RE.source, CORRECTION_RE.flags);
  while ((match = re.exec(content)) !== null) {
    results.push({
      original: match[1],
      corrected: match[2],
      reason: match[3].trim(),
    });
  }
  return results;
}

/**
 * Crée le fichier d'audit si des corrections factuelles ou des régions non-textuelles existent.
 * @param {string} structuredContent - Sortie LLM complète
 * @param {Array<{id:string, x1:number, y1:number, x2:number, y2:number}>} imageRegions
 * @param {string} sourceName - Nom du fichier source (ex: 'IMG_2406.jpeg')
 * @param {string} dateIngestion - ISO timestamp
 * @returns {string|null} Chemin du fichier créé, ou null si non nécessaire
 */
export function writeAuditLog(structuredContent, imageRegions, sourceName, dateIngestion) {
  const corrections = parseCorrections(structuredContent);
  if (corrections.length === 0 && imageRegions.length === 0) return null;

  const baseName = path.basename(sourceName, path.extname(sourceName));
  fs.mkdirSync(AUDIT_DIR, { recursive: true });
  const auditPath = path.join(AUDIT_DIR, `${baseName}.audit.md`);

  const lines = [
    `# Audit — ${baseName}`,
    '',
    `**Date d'ingestion** : ${dateIngestion}`,
    `**Modèle OCR** : Mistral OCR (free)`,
    `**Modèle structuration** : mistral-small-latest`,
    '',
  ];

  if (corrections.length > 0) {
    lines.push('## Corrections factuelles appliquées', '');
    lines.push('| Original (OCR) | Corrigé | Raison |');
    lines.push('|----------------|---------|--------|');
    for (const c of corrections) {
      lines.push(`| "${c.original}" | "${c.corrected}" | ${c.reason} |`);
    }
    lines.push('');
  }

  if (imageRegions.length > 0) {
    lines.push('## Zones non-textuelles détectées', '');
    imageRegions.forEach((r, i) => {
      lines.push(`- Schéma ${i + 1} : \`${r.id}\` — bounding box (${r.x1}, ${r.y1}, ${r.x2}, ${r.y2})`);
    });
    lines.push('');
  }

  fs.writeFileSync(auditPath, lines.join('\n'));
  return auditPath;
}
```

```js
#!/usr/bin/env node
// ingest.js
import 'dotenv/config';
import path from 'node:path';
import { runOCR } from './src/ocr.js';
import { structureContent } from './src/structure.js';
import { buildNote } from './src/format.js';
import { writeAuditLog } from './src/audit.js';

const SUBJECT = process.env.SCRIP_SUBJECT ?? 'histoire';
const LEVEL = process.env.SCRIP_LEVEL ?? '1ère';

async function main() {
  const imagePath = process.argv[2];

  if (!imagePath) {
    console.error('Usage: node ingest.js <chemin_image.jpg>');
    process.exit(1);
  }

  const mistralKey = process.env.MISTRAL_API_KEY;
  const openrouterKey = process.env.OPENROUTER_API_KEY;

  if (!mistralKey || !openrouterKey) {
    console.error('Erreur : MISTRAL_API_KEY et OPENROUTER_API_KEY doivent être définis dans .env');
    process.exit(1);
  }

  const absolutePath = path.resolve(imagePath);
  const sourceName = path.basename(imagePath);

  process.stderr.write(`[1/3] OCR en cours sur ${sourceName}...\n`);
  const { rawMarkdown, imageRegions } = await runOCR(absolutePath, mistralKey);

  process.stderr.write(`[2/3] Structuration via LLM (${SUBJECT})...\n`);
  const structured = await structureContent(rawMarkdown, SUBJECT, LEVEL, openrouterKey, mistralKey);

  process.stderr.write(`[3/3] Assemblage de la note...\n`);
  const now = new Date();
  const dateCours = now.toISOString().split('T')[0];
  const dateIngestion = now.toISOString();

  const note = buildNote(structured, {
    subject: SUBJECT,
    level: LEVEL,
    dateCours,
    dateIngestion,
    sources: [sourceName],
  });

  process.stdout.write(note);

  const auditPath = writeAuditLog(structured, imageRegions, sourceName, dateIngestion);
  if (auditPath) {
    process.stderr.write(`[audit] ${auditPath}\n`);
  }
}

main().catch((err) => {
  console.error('Erreur pipeline :', err.message);
  process.exit(1);
});
```
