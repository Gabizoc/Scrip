# Auth — Login email + mdp + magic link + reset — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Protéger l'app Scrip derrière un login email+mdp avec magic link et reset par email, en partageant le hash du mot de passe avec le WebDAV.

**Architecture:** Un middleware `requireAuth` protège toutes les routes `/api/*` sauf `/api/auth/*`. Le mot de passe est stocké dans `db.data.settings.auth.passwordHash` (bcrypt). Les tokens temporaires (magic, reset, setup) sont stockés en clair dans le db avec expiry. Les emails sont envoyés via nodemailer (config SMTP dans `.env`).

**Tech Stack:** `express-session` (session cookie httpOnly 30j), `nodemailer` (SMTP), `bcryptjs` (déjà installé), `node:crypto` (génération tokens), React Router (pages auth frontend), React Query (status check).

```js
import { JSONFile } from 'lowdb/node';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import fs from 'node:fs';

const __dirname = path.dirname(fileURLToPath(import.meta.url));

const DEFAULT_DATA = {
  settings: {
    theme: 'auto',
    vaultPath: path.join(__dirname, '../../vault'),
    currentLevelId: null,
    publicUrl: '',
    auth: {
      passwordHash:      null,
      pendingSetupToken: null,
      pendingResetToken: null,
      pendingMagicToken: null,
    },
    webdav: {
      enabled:      false,
      username:     'scrip',
      passwordHash: null,  // conservé pour rétrocompat Obsidian, mais non utilisé pour auth
    },
  },
  levels:   [],
  subjects: [],
  notes:    [],
};

const DB_PATH = process.env.DB_PATH
  ?? path.join(__dirname, '../data/db.json');

fs.mkdirSync(path.dirname(DB_PATH), { recursive: true });

const adapter = new JSONFile(DB_PATH);
export const db = new Low(adapter, DEFAULT_DATA);

export async function initDb() {
  await db.read();

  const existing = db.data.settings ?? {};

  db.data = {
    settings: {
      ...DEFAULT_DATA.settings,
      ...existing,
      auth: {
        ...DEFAULT_DATA.settings.auth,
        ...(existing.auth ?? {}),
      },
      webdav: {
        ...DEFAULT_DATA.settings.webdav,
        ...(existing.webdav ?? {}),
      },
    },
    levels:   db.data.levels   ?? [],
    subjects: db.data.subjects ?? [],
    notes:    db.data.notes    ?? [],
  };

  // Migration : si auth.passwordHash est vide mais webdav.passwordHash existe,
  // copier pour que l'ancien mot de passe WebDAV devienne le mot de passe de l'app.
  if (!db.data.settings.auth.passwordHash && db.data.settings.webdav.passwordHash) {
    db.data.settings.auth.passwordHash = db.data.settings.webdav.passwordHash;
  }

  await db.write();
}
```

```js
/**
 * Envoie un email via SMTP (config dans .env).

export async function sendMail({ to, subject, html }) {
  const transporter = nodemailer.createTransport({
    host:   process.env.SMTP_HOST,
    port:   Number(process.env.SMTP_PORT ?? 587),
    secure: process.env.SMTP_SECURE === 'true',
    auth: {
      user: process.env.SMTP_USER,
      pass: process.env.SMTP_PASS,
    },
  });

  await transporter.sendMail({
    from:    process.env.SMTP_FROM,
    to,
    subject,
    html,
  });
}
```

```js
// backend/src/middleware/requireAuth.js

/**
 * Middleware Express : vérifie que la session contient loggedIn === true.
 * Retourne 401 JSON si non authentifié.
 */
export function requireAuth(req, res, next) {
  if (req.session?.loggedIn === true) return next();
  res.status(401).json({ error: 'Non authentifié' });
}
```

```js
import express from 'express';
import cors from 'cors';
import session from 'express-session';
import { errorHandler }          from './middleware/errorHandler.js';
import { requireAuth }           from './middleware/requireAuth.js';
import { settingsRouter }        from './routes/settings.js';
import { levelsRouter }          from './routes/levels.js';
import { subjectsRouter }        from './routes/subjects.js';
import { notesRouter }           from './routes/notes.js';
import { ingestRouter }          from './routes/ingest.js';
import { createWebdavMiddleware } from './webdav.js';

export function createApp() {
  const app = express();

  app.use(cors({
    origin: process.env.NODE_ENV === 'production'
      ? false
      : 'http://localhost:5000',
    credentials: true,
  }));

  app.use(express.json());

  // Session cookie httpOnly — 30 jours
  app.use(session({
    secret:            process.env.SESSION_SECRET ?? 'dev-secret-change-me',
    resave:            false,
    saveUninitialized: false,
    cookie: {
      httpOnly: true,
      secure:   process.env.NODE_ENV === 'production',
      maxAge:   30 * 24 * 60 * 60 * 1000,
    },
  }));

  // Routes auth — PUBLIQUES (avant requireAuth)
  // authRouter sera ajouté à la Task 6

  // Toutes les routes /api/* sont protégées
  app.use('/api', requireAuth);

  app.use('/api/settings', settingsRouter);
  app.use('/api/levels',   levelsRouter);
  app.use('/api/subjects', subjectsRouter);
  app.use('/api/notes',    notesRouter);
  app.use('/api/ingest',   ingestRouter);

  app.use('/webdav', createWebdavMiddleware());

  app.use(errorHandler);
  return app;
}
```

```js
// backend/src/routes/auth.js
import { Router }  from 'express';
import bcrypt      from 'bcryptjs';
import crypto      from 'node:crypto';
import { db }      from '../db.js';
import { sendMail } from '../lib/mailer.js';

export const authRouter = Router();

/** Génère un token aléatoire de 32 octets (64 chars hex) */
function generateToken() {
  return crypto.randomBytes(32).toString('hex');
}

/** Crée un objet token avec expiry 15 minutes */
function makeToken() {
  return { token: generateToken(), expiresAt: Date.now() + 15 * 60 * 1000 };
}

/** Vérifie qu'un objet token est valide (non nul, non expiré, token correct) */
function isValidToken(stored, incoming) {
  if (!stored) return false;
  if (Date.now() > stored.expiresAt) return false;
  return stored.token === incoming;
}

/** URL publique du serveur (depuis db ou env fallback) */
async function getBaseUrl() {
  await db.read();
  return (db.data.settings.publicUrl ?? '').replace(/\/$/, '') || `http://localhost:${process.env.PORT ?? 5001}`;
}

// ── GET /api/auth/status ──────────────────────────────────────────────────────
authRouter.get('/status', async (req, res) => {
  await db.read();
  res.json({
    loggedIn:   req.session?.loggedIn === true,
    needsSetup: db.data.settings.auth.passwordHash === null,
  });
});

// ── POST /api/auth/setup-request ─────────────────────────────────────────────
authRouter.post('/setup-request', async (req, res) => {
  await db.read();
  if (db.data.settings.auth.passwordHash !== null) {
    return res.status(403).json({ error: 'Setup déjà effectué' });
  }

  const pending = makeToken();
  db.data.settings.auth.pendingSetupToken = pending;
  await db.write();

  const baseUrl = await getBaseUrl();
  const link    = `${baseUrl}/setup/password?token=${pending.token}`;

  try {
    await sendMail({
      to:      process.env.AUTH_EMAIL,
      subject: 'Scrip — Configuration initiale',
      html:    `<p>Cliquez sur le lien pour définir votre mot de passe :</p>
                <p><a href="${link}">${link}</a></p>
                <p>Ce lien expire dans 15 minutes.</p>`,
    });
  } catch (err) {
    console.error('[auth] Erreur envoi email setup:', err.message);
    // On continue — le token est en db, on peut réessayer
  }

  res.json({ ok: true });
});

// ── POST /api/auth/setup-complete ────────────────────────────────────────────
authRouter.post('/setup-complete', async (req, res) => {
  await db.read();
  if (db.data.settings.auth.passwordHash !== null) {
    return res.status(403).json({ error: 'Setup déjà effectué' });
  }

  const { token, password } = req.body ?? {};
  if (!token || !password) {
    return res.status(400).json({ error: 'token et password requis' });
  }
  if (typeof password !== 'string' || password.length < 8) {
    return res.status(400).json({ error: 'Mot de passe trop court (8 caractères minimum)' });
  }
  if (!isValidToken(db.data.settings.auth.pendingSetupToken, token)) {
    return res.status(401).json({ error: 'Token invalide ou expiré' });
  }

  db.data.settings.auth.passwordHash      = await bcrypt.hash(password, 10);
  db.data.settings.auth.pendingSetupToken = null;
  // Synchroniser avec webdav.passwordHash pour Obsidian
  db.data.settings.webdav.passwordHash    = db.data.settings.auth.passwordHash;
  await db.write();

  req.session.loggedIn = true;
  res.json({ ok: true });
});

// ── POST /api/auth/login ──────────────────────────────────────────────────────
authRouter.post('/login', async (req, res) => {
  await db.read();
  const { email, password } = req.body ?? {};

  const emailOk = typeof email === 'string'
    && email.toLowerCase() === process.env.AUTH_EMAIL.toLowerCase();

  const hash       = db.data.settings.auth.passwordHash;
  const passwordOk = hash ? await bcrypt.compare(String(password ?? ''), hash) : false;

  if (!emailOk || !passwordOk) {
    return res.status(401).json({ error: 'Email ou mot de passe incorrect' });
  }

  req.session.loggedIn = true;
  res.json({ ok: true });
});

// ── POST /api/auth/magic ──────────────────────────────────────────────────────
authRouter.post('/magic', async (req, res) => {
  await db.read();
  if (db.data.settings.auth.passwordHash === null) {
    return res.status(403).json({ error: 'Setup non effectué' });
  }

  const pending = makeToken();
  db.data.settings.auth.pendingMagicToken = pending;
  await db.write();
```

```js
// backend/src/webdav.js
import webdavPkg from 'webdav-server';
import bcrypt    from 'bcryptjs';
import { db }   from './db.js';

const { v2: webdav } = webdavPkg;

const BYPASS_USER = 'scrip-internal';
const BYPASS_PASS = 'bypass-' + Math.random().toString(36).slice(2);

const _userManager = new webdav.SimpleUserManager();
_userManager.addUser(BYPASS_USER, BYPASS_PASS, true);

let _server    = null;
let _vaultPath = null;

async function getWebdavServer(vaultPath) {
  if (_server && _vaultPath === vaultPath) return _server;

  const server = new webdav.WebDAVServer({
    httpAuthentication: new webdav.HTTPBasicAuthentication(_userManager, 'Scrip'),
  });

  await new Promise((resolve, reject) => {
    server.setFileSystem('/', new webdav.PhysicalFileSystem(vaultPath), (success) => {
      if (success) resolve();
      else reject(new Error(`WebDAV: impossible de monter le filesystem ${vaultPath}`));
    });
  });

  _server    = server;
  _vaultPath = vaultPath;
  return server;
}

export function createWebdavMiddleware() {
  return async (req, res, next) => {
    try {
      await db.read();
      const { webdav: cfg, auth, vaultPath } = db.data.settings;

      if (!vaultPath) {
        return res.status(503).json({ error: 'Vault path non configuré dans les paramètres Scrip' });
      }
      if (!cfg?.enabled) {
        return res.status(503).json({ error: 'WebDAV désactivé dans les paramètres Scrip' });
      }

      const authHeader = req.headers['authorization'] ?? '';
      if (!authHeader.startsWith('Basic ')) {
        res.setHeader('WWW-Authenticate', 'Basic realm="Scrip"');
        return res.status(401).end('Authentification requise');
      }

      const decoded  = Buffer.from(authHeader.slice(6), 'base64').toString('utf8');
      const colonIdx = decoded.indexOf(':');
      if (colonIdx === -1) {
        res.setHeader('WWW-Authenticate', 'Basic realm="Scrip"');
        return res.status(401).end('Authentification requise');
      }

      const username = decoded.slice(0, colonIdx);
      const password = decoded.slice(colonIdx + 1);

      const usernameOk = cfg.username === username;
      // Utilise auth.passwordHash (source unique) — synchronisé avec webdav.passwordHash
      const passwordOk = auth.passwordHash
        ? await bcrypt.compare(password, auth.passwordHash)
        : false;

      if (!usernameOk || !passwordOk) {
        res.setHeader('WWW-Authenticate', 'Basic realm="Scrip"');
        return res.status(401).end('Identifiants incorrects');
      }

      req.headers['authorization'] =
        `Basic ${Buffer.from(`${BYPASS_USER}:${BYPASS_PASS}`).toString('base64')}`;

      const server = await getWebdavServer(vaultPath);
      webdav.extensions.express('/', server)(req, res, next);
    } catch (err) {
      next(err);
    }
  };
}
```

```js
// frontend/src/lib/auth.js

async function authRequest(method, path, body) {
  const opts = {
    method,
    credentials: 'include',
    headers: body ? { 'Content-Type': 'application/json' } : {},
    body:    body ? JSON.stringify(body) : undefined,
  };
  const res = await fetch(`/api/auth${path}`, opts);
  if (res.status === 204) return null;
  const data = await res.json().catch(() => ({ error: res.statusText }));
  if (!res.ok) throw new Error(data.error ?? res.statusText);
  return data;
}

export const authApi = {

  magic:          ()                       => authRequest('POST', '/magic'),
  forgot:         ()                       => authRequest('POST', '/forgot'),
  reset:          (token, password)        => authRequest('POST', '/reset',          { token, password }),
  setupRequest:   ()                       => authRequest('POST', '/setup-request'),
```

```jsx
import { useState } from 'react';
import { useNavigate, useSearchParams } from 'react-router-dom';
import { authApi } from '../lib/auth.js';

export default function SetupPassword() {
  const navigate             = useNavigate();
  const [searchParams]       = useSearchParams();
  const token                = searchParams.get('token') ?? '';

  const [password,  setPassword]  = useState('');
  const [confirm,   setConfirm]   = useState('');
  const [error,     setError]     = useState('');
  const [loading,   setLoading]   = useState(false);

  async function handleSubmit(e) {
    e.preventDefault();
    setError('');
    if (password !== confirm) {
      setError('Les mots de passe ne correspondent pas.');
      return;
    }
    if (password.length < 8) {
      setError('Le mot de passe doit faire au moins 8 caractères.');
      return;
    }
    setLoading(true);
    try {
      await authApi.setupComplete(token, password);
      navigate('/', { replace: true });
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  }

  if (!token) {
    return (
      <div className="min-h-screen bg-bg-base flex items-center justify-center p-4">
        <p className="text-status-danger text-sm">Lien invalide. Retournez sur <a href="/setup" className="underline">la page de setup</a>.</p>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-bg-base flex items-center justify-center p-4">
      <div className="w-full max-w-sm space-y-6">
        <div className="text-center">
          <h1 className="text-2xl font-bold text-text-primary">Choisir un mot de passe</h1>
          <p className="text-sm text-text-secondary mt-1">Ce mot de passe protège l'app et le WebDAV.</p>
```

```jsx
import { useNavigate, useParams } from 'react-router-dom';

  const navigate       = useNavigate();
  const { token }      = useParams();

  const [password,  setPassword]  = useState('');
  const [confirm,   setConfirm]   = useState('');
  const [error,     setError]     = useState('');
  const [loading,   setLoading]   = useState(false);
  const [done,      setDone]      = useState(false);
```

```jsx
// frontend/src/components/AuthGuard.jsx
import { useQuery } from '@tanstack/react-query';
import { Navigate } from 'react-router-dom';
import { authApi } from '../lib/auth.js';

/**
 * Protège les routes enfants.
 * - needsSetup → /setup
 * - non authentifié → /login
 * - authentifié → render children
 */
export default function AuthGuard({ children }) {
  const { data, isLoading, isError } = useQuery({
    queryKey:           ['auth-status'],
    queryFn:            authApi.status,
    retry:              false,
    staleTime:          60_000,  // revalide toutes les minutes
  });

  if (isLoading) {
    // Écran blanc pendant la vérification — évite le flash de la page login
    return <div className="min-h-screen bg-bg-base" />;
  }

  if (isError || !data?.loggedIn) {
    if (data?.needsSetup) return <Navigate to="/setup" replace />;
    return <Navigate to="/login" replace />;
  }

  return children;
}
```

```jsx
import { Routes, Route, useLocation } from 'react-router-dom';
import { SubjectsProvider } from './lib/SubjectsContext.jsx';
import Sidebar        from './components/Sidebar.jsx';
import BottomNav      from './components/BottomNav.jsx';
import AuthGuard      from './components/AuthGuard.jsx';
import Home           from './pages/Home.jsx';
import Capture        from './pages/Capture.jsx';
import Preview        from './pages/Preview.jsx';
import Settings       from './pages/Settings/index.jsx';
import LevelsManager  from './pages/Settings/LevelsManager.jsx';
import SubjectsManager from './pages/Settings/SubjectsManager.jsx';
import Login          from './pages/Login.jsx';
import Setup          from './pages/Setup.jsx';
import SetupPassword  from './pages/SetupPassword.jsx';
import ResetPassword  from './pages/ResetPassword.jsx';

function Background() {
  return <div className="fixed inset-0 -z-10 pointer-events-none bg-bg-base" aria-hidden="true" />;
}

function PageWrapper({ children }) {
  const location = useLocation();
  return <div key={location.pathname} className="page-enter">{children}</div>;
}

function ProtectedLayout() {
  return (
    <SubjectsProvider>
      <div className="min-h-screen bg-bg-base text-text-primary">
        <Background />
        <Sidebar />
        <BottomNav />
        <div className="md:pl-56 min-h-screen pb-16 md:pb-0">

            <Routes>

              <Route path="/settings/subjects"  element={<SubjectsManager />} />
            </Routes>

        </div>

      <Route path="*" element={

}
```
